# Portable build entry for the exact vendored compiler sources/shims.
# This entry point replaces environment-specific library/include paths only.
cmake_minimum_required(VERSION 3.20)
project(MoreBratsCaprica LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MB_BOOST_ROOT "" CACHE PATH "Boost 1.83.0 header tree or installation prefix")
find_path(MB_BOOST_INCLUDE boost/version.hpp HINTS "${MB_BOOST_ROOT}" "${MB_BOOST_ROOT}/include")
find_library(MB_BOOST_FILESYSTEM NAMES boost_filesystem libboost_filesystem.so.1.83.0)
find_library(MB_BOOST_PROGRAM_OPTIONS NAMES boost_program_options libboost_program_options.so.1.83.0)
if(NOT MB_BOOST_INCLUDE OR NOT MB_BOOST_FILESYSTEM OR NOT MB_BOOST_PROGRAM_OPTIONS)
  message(FATAL_ERROR "Boost 1.83.0 headers and filesystem/program_options libraries are required. Set MB_BOOST_ROOT if headers are outside the normal compiler search path.")
endif()
file(STRINGS "${MB_BOOST_INCLUDE}/boost/version.hpp" MB_BOOST_VERSION REGEX "^#define BOOST_VERSION ")
if(NOT MB_BOOST_VERSION MATCHES "108300")
  message(FATAL_ERROR "This preserved toolchain was tested with Boost 1.83.0 (108300).")
endif()
find_package(Threads REQUIRED)
file(GLOB_RECURSE CAPRICA_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/caprica/Caprica/*.cpp")
add_library(MBPugixml STATIC "${CMAKE_CURRENT_SOURCE_DIR}/../ThirdParty/pugixml/src/pugixml.cpp")
target_include_directories(MBPugixml PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../ThirdParty/pugixml/src")
add_executable(Caprica ${CAPRICA_SOURCES})
target_include_directories(Caprica PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/compat" "${CMAKE_CURRENT_SOURCE_DIR}/caprica/Caprica" "${MB_BOOST_INCLUDE}")
target_compile_options(Caprica PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/compat/compat.h" -msse4.2 -fpermissive -Wno-deprecated-declarations)
target_link_libraries(Caprica PRIVATE "${MB_BOOST_FILESYSTEM}" "${MB_BOOST_PROGRAM_OPTIONS}" MBPugixml Threads::Threads)
